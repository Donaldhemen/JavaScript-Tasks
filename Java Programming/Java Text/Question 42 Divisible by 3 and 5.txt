Question 42 Divisible by 3 and 5

// read an integer
// if divisible by 3 and 5
// if divisible by 3 only
// if divisible by 5 only
// else divisible neither 

	public static void main(String[] args) {
	Scanner input = new Scanner(System.in);
	
	System.out.print("Enter integer : ");
	int number = input.nextInt();

	if (number % 3 == 0 && number % 5 == 0) {
	System.out.print("Divisible by 3 and 5");
	}
	else if (number % 3 == 0) {
	System.out.print("Divisible by 3 only");
	}
	else if (number % 5 == 0) {
	System.out.print("Divisible by 5 only");
	}
	else {
	System.out.print("Divisible by neither");
	}
