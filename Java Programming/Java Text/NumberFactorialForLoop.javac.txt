public class NumberFactorialForLoop {
    public static void main(String[] args) {
	Scanner input = new Scanner(System.in);
	
	System.out.print("Enter number: ");
	int counter = input.nextInt();
	int factorial = 1;
	

		while (counter >= 1) { 
			factorial *= counter;
			
			counter-- ;
		}
		System.out.printf("%d ", factorial);
	}
}
