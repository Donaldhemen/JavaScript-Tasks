// if income  <= 300000 tax:0%
// if income  <= 600000 tax:7%

	System.out.print("Enter integer between 1 and 7: ");
	double income = input.nextDouble();

	if (income <= 300000) {
	double totalTax = income * 0 / 100;
	
	System.out.printf("Total tax owed is: %f", totalTax);
	}
	else if (income <= 600000) {
	double totalTax = income * 7 / 100;
	
	System.out.printf("Total tax owed is: %f", totalTax);
	}
	else  {
	double totalTax = income * 15 / 100;
	
	System.out.printf("Total tax owed is: %f", totalTax);
	}
